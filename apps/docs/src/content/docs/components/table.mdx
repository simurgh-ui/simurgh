Table keeps native caption, section, row, header, and cell semantics while adding stable styling hooks.

<ComponentPreview title="Table">
  <table data-slot="table">
    <caption data-slot="table-caption">Recent releases</caption>
    <thead data-slot="table-header">
      <tr data-slot="table-row">
        <th scope="col" data-slot="table-head">
          Version
        </th>
        <th scope="col" data-slot="table-head">
          Status
        </th>
      </tr>
    </thead>
    <tbody data-slot="table-body">
      <tr data-slot="table-row">
        <td data-slot="table-cell">0.1.0</td>
        <td data-slot="table-cell">Stable</td>
      </tr>
      <tr data-slot="table-row">

        <td data-slot="table-cell">Planned</td>

  </table>
