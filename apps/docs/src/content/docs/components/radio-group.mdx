import { Tabs, TabItem } from '@astrojs/starlight/components';
import ComponentPreview from '../../../components/ComponentPreview.astro';

Radio Group allows one value from a visible set. Arrow keys move and select, Home/End jump to the boundaries, horizontal movement follows RTL direction, and the selected value participates in native form submission.

    </label>
    <label class="preview-row">

  </fieldset>
</ComponentPreview>
